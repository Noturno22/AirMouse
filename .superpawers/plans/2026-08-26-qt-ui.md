# Qt UI Implementation Plan — Mãouse

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpawers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the OpenCV cv2.imshow overlay with a PySide6 Qt application that displays the camera feed, hand tracking overlays, gesture badges, status indicators, and settings — all styled according to the IDENTIDADE_VISUAL.md design system.

**Architecture:** PySide6 application with a QTimer (0ms, process-as-fast-as-possible) polling the core processing state each tick. Camera feed displayed via QLabel/QPixmap. Hand skeleton painted via QPainter overlay on a custom QWidget. All UI widgets composed in a frameless MainWindow. QSystemTrayIcon replaces pystray. Theme module centralizes all colors/fonts from IDENTIDADE_VISUAL.md. Backward-compatible: `--ui qt` flag enables Qt UI, falls back to OpenCV if PySide6 unavailable.

**Tech Stack:** PySide6, numpy, OpenCV (camera + frame conversion only), pynput (existing)

```python
"""Camera feed display with hand skeleton overlay."""
import cv2
import numpy as np

from PySide6.QtCore import Qt, QRect, QPoint
from PySide6.QtGui import (
    QImage, QPixmap, QPainter, QPen, QColor, QBrush, QPolygon,
)
from PySide6.QtWidgets import QWidget

from core.tracker import HAND_CONNECTIONS
from core.gestures import Gesture
from ui.theme import (
    GESTURE_COLORS, TEXT_SECONDARY, BG_PRIMARY,
    gesture_color, gesture_label, FONT_MONO,
)


class CameraView(QWidget):
    """Displays the camera feed and paints hand skeleton overlay."""

    def __init__(self, parent=None):
        super().__init__(parent)
        self._pixmap = QPixmap()
        self._scale = 1.0
        self._offset = QPoint(0, 0)
        self._all_frames = {}
        self._active_side = None
        self._flash = 0
        self.setMinimumSize(320, 240)

    def update_frame(self, frame_bgr, all_frames, active_side, flash):
        """Update the displayed frame and overlay data.
        
        Args:
            frame_bgr: numpy BGR array from camera
            all_frames: dict {side: HandFrame}
            active_side: "Left" or "Right" or None
            flash: int, green flash counter (0 = no flash)
        """
        if frame_bgr is None:
            return
        rgb = cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB)
        h, w, ch = rgb.shape
        bytes_per_line = ch * w
        qimg = QImage(rgb.data, w, h, bytes_per_line, QImage.Format_RGB888).copy()
        self._pixmap = QPixmap.fromImage(qimg)
        self._all_frames = all_frames or {}
        self._active_side = active_side
        self._flash = flash
        self._recalculate_layout()
        self.update()

    def _recalculate_layout(self):
        """Calculate scale and offset to fit pixmap in widget, keeping aspect ratio."""
        if self._pixmap.isNull():
            return
        pw, ph = self._pixmap.width(), self._pixmap.height()
        ww, wh = self.width(), self.height()
        self._scale = min(ww / pw, wh / ph) if pw and ph else 1.0
        sw, sh = int(pw * self._scale), int(ph * self._scale)
        self._offset = QPoint((ww - sw) // 2, (wh - sh) // 2)

    def resizeEvent(self, event):
        super().resizeEvent(event)
        self._recalculate_layout()

    def paintEvent(self, event):
        painter = QPainter(self)
        painter.setRenderHint(QPainter.Antialiasing)

        # Draw camera feed
        if not self._pixmap.isNull():
            src_rect = QRect(0, 0, self._pixmap.width(), self._pixmap.height())
            dst_rect = QRect(self._offset, self._pixmap.size() * self._scale)
            painter.drawPixmap(dst_rect, self._pixmap, src_rect)
        else:
            painter.fillRect(self.rect(), BG_PRIMARY)

        # Draw hand overlays
        if self._all_frames:
            self._paint_hands(painter)

        # Draw flash overlay
        if self._flash > 0:
            painter.fillRect(self.rect(), QColor(90, 220, 90, 40))
            pen = QPen(QColor(90, 220, 90), 6)
            painter.setPen(pen)
            painter.drawRect(3, 3, self.width() - 6, self.height() - 6)

        painter.end()

    def _paint_hands(self, painter):
        """Draw all detected hands with skeleton, crosshair, and gesture indicators."""
        for side, hf in self._all_frames.items():
            color = gesture_color(hf.gesture)

            if side != self._active_side:
                # Non-active hands: dim gray skeleton
                self._draw_hand_skeleton(painter, hf, TEXT_SECONDARY, alpha=120)
            else:
                # Active hand: full color
                self._draw_hand_skeleton(painter, hf, color, alpha=255)
                self._draw_palm_crosshair(painter, hf, color)

                # Pinch line (thumb to index)
                if hf.gesture == Gesture.PINCH:
                    thumb = self._to_widget(hf.points_px[4])
                    tip = self._to_widget(hf.index_tip)
                    pen = QPen(QColor(90, 220, 90), 2)
                    painter.setPen(pen)
                    painter.drawLine(thumb, tip)

                # Peace indicator: circle on middle finger tip
                if hf.gesture == Gesture.PEACE:
                    mid = self._to_widget(hf.points_px[12])
                    pen = QPen(QColor(255, 80, 200), 2)
                    painter.setPen(pen)
                    painter.setBrush(Qt.NoBrush)
                    painter.drawEllipse(mid, 7, 7)

                # Pinch ratio bar at bottom
                self._draw_pinch_bar(painter, hf)

    def _draw_hand_skeleton(self, painter, hf, color, alpha=255):
        """Draw lines connecting landmark points."""
        c = QColor(color)
        c.setAlpha(alpha)
        pen = QPen(c, 1)
        painter.setPen(pen)
        for a, b in HAND_CONNECTIONS:
            pa = self._to_widget(hf.points_px[a])
            pb = self._to_widget(hf.points_px[b])
            painter.drawLine(pa, pb)

    def _draw_palm_crosshair(self, painter, hf, color):
        """Draw crosshair and circle at palm center."""
        px, py = self._to_widget(hf.palm_center)
        pen = QPen(color, 2)
        painter.setPen(pen)
        painter.drawLine(px - 14, py, px + 14, py)
        painter.drawLine(px, py - 14, px, py + 14)
        painter.setBrush(Qt.NoBrush)
        painter.drawEllipse(QPoint(px, py), 9, 9)

    def _draw_pinch_bar(self, painter, hf):
        """Draw horizontal pinch ratio bar at the bottom."""
        w, h = self.width(), self.height()
        ratio = min(hf.pinch_ratio / 1.2, 1.0)
        bar_width = int(ratio * (w - 24))
        color = gesture_color(hf.gesture) if hf.gesture == Gesture.PINCH else TEXT_SECONDARY
        painter.setPen(Qt.NoPen)
        painter.setBrush(QBrush(color))
        painter.drawRect(12, h - 30, bar_width, 6)

    def _to_widget(self, point):
        """Convert a frame-coordinate point (x, y) to widget coordinates."""
        if self._pixmap.isNull():
            return QPoint(int(point[0]), int(point[1]))
        x = int(point[0] * self._scale + self._offset.x())
        y = int(point[1] * self._scale + self._offset.y())
        return QPoint(x, y)
```

```python
        self.setObjectName("GestureBadge")
        self.setFont(FONT_BADGE)
        self.setAlignment(Qt.AlignCenter)

        self.setMinimumWidth(180)

            label = "PAUSA"

            label = gesture_label(gesture)

        self.setText(text)
        self._current_color = color
        self.setStyleSheet(

            f"border: 2px solid {color.name()};"
            f"border-radius: 6px;"
            f"color: {color.name()};"
```

```python
"""Top-right stacked status badges: AI confidence, magnifier, hand count."""
from PySide6.QtCore import Qt
from PySide6.QtWidgets import QWidget, QHBoxLayout, QLabel

from ui.theme import (
    BG_OVERLAY, TEXT_PRIMARY, TEXT_SECONDARY, SUCCESS, FONT_MONO,
)


class StatusBadges(QWidget):
    """Horizontal stack of status badges (AI, magnifier, hands)."""

    def __init__(self, parent=None):
        super().__init__(parent)
        layout = QHBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(6)
        layout.addStretch()

        self._ai_badge = self._make_badge("IA  ok")
        self._mag_badge = self._make_badge("")
        self._hands_badge = self._make_badge("")

        layout.addWidget(self._ai_badge)
        layout.addWidget(self._mag_badge)
        layout.addWidget(self._hands_badge)

        self._mag_badge.hide()
        self._hands_badge.hide()

    def _make_badge(self, text):
        lbl = QLabel(text)
        lbl.setObjectName("StatusBadge")
        lbl.setFont(FONT_MONO)
        lbl.setAlignment(Qt.AlignCenter)
        lbl.setFixedHeight(28)
        lbl.setMinimumWidth(60)
        return lbl

    def update_state(self, ai_on, ai_conf, magnify, hands):
        """Update all status badges.
        
        Args:
            ai_on: bool - AI classifier active
            ai_conf: float 0.0-1.0 - AI confidence
            magnify: str - magnifier action label or ""
            hands: int - number of detected hands (0-2)
        """
        # AI badge
        if ai_on:
            self._ai_badge.show()
            txt = f"IA {int(ai_conf * 100):3d}%" if hands > 0 else "IA  ok"
            color = SUCCESS if ai_conf >= 0.72 else TEXT_PRIMARY
            self._ai_badge.setText(txt)
            self._ai_badge.setStyleSheet(
                f"background-color: rgba(10,10,18,204);"
                f"border: 1px solid {color.name()};"
                f"border-radius: 4px; color: {color.name()};"
                f"padding: 4px 10px;"
            )
        else:
            self._ai_badge.hide()

        # Magnifier badge
        if magnify:
            self._mag_badge.show()
            self._mag_badge.setText(f"LUPA {magnify}")
            self._mag_badge.setStyleSheet(
                f"background-color: rgba(10,10,18,204);"
                f"border: 1px solid #FF50C8; border-radius: 4px;"
                f"color: #FF50C8; padding: 4px 10px;"
            )
        else:
            self._mag_badge.hide()

        # Hands badge
        if hands > 0:
            self._hands_badge.show()
            self._hands_badge.setText(f"{hands} MAOS")
            self._hands_badge.setStyleSheet(
                f"background-color: rgba(10,10,18,204);"
                f"border: 1px solid {TEXT_PRIMARY.name()}; border-radius: 4px;"
                f"color: {TEXT_PRIMARY.name()}; padding: 4px 10px;"
            )
        else:
            self._hands_badge.hide()
```

```python
    # ── Processing Loop ─────────────────────────────────────────────
    def _tick(self):
        """Main processing tick — called by QTimer(0)."""
        frame, seq = self._cam.read()
        if frame is None:
            return

        if self._warmup > 0:
            self._warmup -= 1
            return

        if self._cfg.mirror:
            frame = cv2.flip(frame, 1)
        h, w = frame.shape[:2]

        # Low-light boost
        if self._light is not None:
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            gmean = float(cv2.mean(gray)[0])
            evt = self._light.feed(gmean)
            if evt:
                self._toast.show_toast(
                    "LUZ BAIXA: realce ativado" if evt == "on" else "LUZ NORMAL"
                )
        if self._light is not None and self._light.active:
            lab = cv2.cvtColor(frame, cv2.COLOR_BGR2LAB)
            lch, ach, bch = cv2.split(lab)
            lch = self._clahe.apply(lch)
            frame = cv2.cvtColor(cv2.merge((lch, ach, bch)), cv2.COLOR_LAB2BGR)

        # Hand tracking
        ts_ms = time.monotonic_ns() // 1_000_000
        rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        hands, sides = self._tracker.process(rgb, ts_ms)
        results = self._pool.update(hands, sides, w, h)
        now = time.perf_counter()

        event = None
        ev_value = None
        hand_frame = None

        if results:
            if self._active_side in results:
                hand_frame, event, ev_value = results[self._active_side]
            else:
                self._active_side = "Right" if "Right" in results else next(iter(results))
                hand_frame, event, ev_value = results[self._active_side]
                self._filters.reset()
                self._last_palm = None
                self._prev_filtered = None
                self._jump_streak = 0
                self._fast_until = 0.0
                self._emitter.clear()

        all_frames = {s: r[0] for s, r in results.items()}

        # Process hand movement
        if hand_frame is not None and hand_frame.gesture != Gesture.NONE:
            palm = hand_frame.palm_center
            accept = True
            if self._last_palm is not None:
                d = math.hypot(palm[0] - self._last_palm[0], palm[1] - self._last_palm[1])
                limit = self._cfg.max_jump_frac * w
                if d <= limit:
                    self._jump_streak = 0
                    self._fast_until = 0.0
                elif now < self._fast_until:
                    pass
                elif self._jump_streak < 2:
                    self._jump_streak += 1
                    accept = False
                else:
                    self._jump_streak = 0
                    self._fast_until = now + 0.5

            if accept:
                if self._last_accept_t is not None:
                    inst_dt = now - self._last_accept_t
                    self._dt_ema = self._dt_ema * 0.7 + min(max(inst_dt, 0.008), 0.2) * 0.3
                self._last_accept_t = now
                self._last_palm = palm
                self._last_hand_t = now
                fx, fy = self._filters.filter(*palm)
                if self._prev_filtered is None:
                    self._prev_filtered = (fx, fy)
                dx = fx - self._prev_filtered[0]
                dy = fy - self._prev_filtered[1]
                self._prev_filtered = (fx, fy)

                gain = self._cfg.move_gain * self._curve.apply(self._filters.vx, self._filters.vy)
                sx = self._mouse.screen_w / w
                sy = self._mouse.screen_h / h
                mdx = dx * gain * sx
                mdy = dy * gain * sy
                movable = (
                    not self._paused
                    and now >= self._freeze_until
                    and hand_frame.gesture in frozenset({Gesture.OPEN, Gesture.ONE, Gesture.PINCH, Gesture.FIST})
                    and not (self._magnifier is not None and self._magnifier.on)
                )
                if movable:
                    emitter_push = True
                    if abs(mdx) < self._cfg.deadzone_px:
                        mdx = 0.0
                    if abs(mdy) < self._cfg.deadzone_px:
                        mdy = 0.0
                    if abs(mdx) > 0 or abs(mdy) > 0:
                        lx, ly = lead_offset(self._filters.vx, self._filters.vy, self._cfg.predict_ms)
                        try:
                            cx, cy = self._mouse.mouse.position
                            pxl, pyl = self._snap.pull((cx, cy), now) if self._snap else (0.0, 0.0)
                        except Exception:
                            pxl = pyl = 0.0
                        self._emitter.push(mdx + lx + pxl, mdy + ly + pyl, self._dt_ema)

                self._tuner.feed(True, self._filters, mdx, mdy)
        else:
            if hand_frame is None:
                self._active_side = None
            self._filters.reset()
            self._last_palm = None
            self._prev_filtered = None
            self._jump_streak = 0
            self._fast_until = 0.0
            self._last_accept_t = None
            self._emitter.clear()
            self._tuner.feed(False, self._filters, 0.0, 0.0)
            if self._button_down and (self._last_hand_t is None or now - self._last_hand_t > 0.20):
                self._mouse.release_left()
                self._button_down = False

        # Handle events
        if event and (not self._paused or event == "left_up"):
            self._handle_event(event, ev_value, now)

        # Handle voice commands
        if self._voice:
            while True:
                try:
                    cmd = self._voice.cmd_queue.get_nowait()
                except Exception:
                    cmd = None
                if cmd is None:
                    break
                self._handle_voice_command(cmd)

        # Auto-tune
        tune_note = self._tuner.maybe_apply(time.monotonic(), self._filters, self._cfg)
        if tune_note:
            self._toast.show_toast(tune_note)
            self._smooth_name = "AUTO"

        # Update FPS
        self._frame_count += 1
        elapsed = time.perf_counter() - self._fps_time
        if elapsed >= 0.5:
            self._fps = self._frame_count / elapsed
            self._frame_count = 0
            self._fps_time = time.perf_counter()

        if self._flash > 0:
            self._flash -= 1

        # Update UI
        self._update_ui(all_frames, hand_frame)

    def _handle_event(self, event, value, now):
        """Handle a gesture event."""
        from main import _handle_media_event, _keyboard_shortcut, _click_assist

        if event == "left_down":

            self._mouse.press_left()
            self._button_down = True
            self._freeze_until = now + self._cfg.click_freeze_ms / 1000.0
            self._flash = 5
            self._emitter.clear()
        elif event == "left_up":
            self._mouse.release_left()
            self._button_down = False
            self._emitter.clear()
        elif event == "right_click":
            self._mouse.right_click()
            self._freeze_until = now + self._cfg.click_freeze_ms / 1000.0
            self._flash = 5
            self._emitter.clear()
        elif event == "copy":
            _keyboard_shortcut("ctrl+c")
            self._toast.show_toast("COPIAR (Ctrl+C)")
            self._freeze_until = now + self._cfg.click_freeze_ms / 1000.0

        elif event == "paste":
            _keyboard_shortcut("ctrl+v")
            self._toast.show_toast("COLAR (Ctrl+V)")
            self._freeze_until = now + self._cfg.click_freeze_ms / 1000.0
            self._flash = 5
        elif event == "scroll" and value is not None:
            self._mouse.scroll(value * self._cfg.scroll_gain_factor)
        else:
            media_note = _handle_media_event(event, value)
            if media_note:
```

## Notes

### Architecture Decisions

1. **QTimer(0) instead of dedicated thread**: The core processing runs in the Qt event loop via QTimer(0). This avoids thread-safety issues with shared state (filters, emitter, etc.) which are NOT thread-safe. The camera read blocks naturally when no frame is available.

2. **Widget composition over custom drawing**: Each HUD element (badge, indicators, voice bar) is a separate QWidget/QLabel with QSS styling. This is simpler than custom painting and allows easy theming.

3. **CameraView custom painting only for overlay**: The hand skeleton overlay uses QPainter because it needs to draw arbitrary lines/shapes on top of the camera feed. The camera feed itself is drawn as a QPixmap.

4. **Backward compatibility**: `--ui opencv` preserves the original OpenCV UI. PySide6 import is lazy — if unavailable, falls back to OpenCV automatically.
